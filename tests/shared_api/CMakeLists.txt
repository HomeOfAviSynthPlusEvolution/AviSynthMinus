cmake_minimum_required(VERSION 3.24)
project(AviSynthSharedApiTests LANGUAGES CXX)
set(AVS_INSTALL_PREFIX "" CACHE PATH "Installed shared AviSynth prefix")
if(NOT AVS_INSTALL_PREFIX)
  message(FATAL_ERROR "Set AVS_INSTALL_PREFIX to an installed shared core")
endif()
find_path(AVS_INCLUDE_DIR avisynth.h PATHS "${AVS_INSTALL_PREFIX}/include/avisynth" NO_DEFAULT_PATH REQUIRED)
find_library(AVS_LIBRARY NAMES avisynth AviSynth PATHS "${AVS_INSTALL_PREFIX}/lib" "${AVS_INSTALL_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
include(FetchContent)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(googletest)
add_executable(shared_api_tests linkage.cpp linkage_tests.cpp
  ../audio_filter/convert_audio_api_tests.cpp
  ../convert_bits_api/convert_bits_api_tests.cpp
  ../composite_api/composite_api_tests.cpp
  ../resize_filter/resize_api_tests.cpp)
add_library(shared_linkage_probe MODULE linkage_probe.cpp)
target_include_directories(shared_linkage_probe PRIVATE "${AVS_INCLUDE_DIR}")
target_compile_features(shared_linkage_probe PRIVATE cxx_std_17)
add_dependencies(shared_api_tests shared_linkage_probe)
target_compile_definitions(shared_api_tests PRIVATE SHARED_LINKAGE_PLUGIN="$<TARGET_FILE:shared_linkage_probe>")
target_compile_features(shared_api_tests PRIVATE cxx_std_17)
target_compile_definitions(shared_api_tests PRIVATE DEPTH_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/video_convert/tests/fixtures/depth/upstream-c.txt")
target_include_directories(shared_api_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/.." "${AVS_INCLUDE_DIR}")
target_link_libraries(shared_api_tests PRIVATE "${AVS_LIBRARY}" GTest::gtest_main)
option(AVS_TEST_IRIS "Test the installed core's internal IrisExpr filter" OFF)
option(AVS_TEST_IRIS_LLVM "Exercise the LLVM backend in Iris host tests" ON)
option(AVS_TEST_IRIS_SLEEF "Exercise the SLEEF backends in Iris host tests" ON)
if(AVS_TEST_IRIS)
  target_sources(shared_api_tests PRIVATE
    ../expr_filter/iris_filter_tests.cpp
    ../expr_filter/expr_filter_tests.cpp
    ../expr_filter/expr_boundary_tests.cpp)
  target_compile_definitions(shared_api_tests PRIVATE AVS_TEST_IRIS_SHARED=1)
  if(AVS_TEST_IRIS_LLVM)
    target_compile_definitions(shared_api_tests PRIVATE AVS_TEST_IRIS_LLVM=1)
  endif()
  if(AVS_TEST_IRIS_SLEEF AND AVS_TEST_IRIS_LLVM)
    target_compile_definitions(shared_api_tests PRIVATE AVS_TEST_IRIS_SLEEF=1)
  endif()
endif()
if(WIN32)
  find_file(AVS_DLL NAMES AviSynth.dll avisynth.dll PATHS "${AVS_INSTALL_PREFIX}/bin" NO_DEFAULT_PATH REQUIRED)
  add_custom_command(TARGET shared_api_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${AVS_DLL}" "$<TARGET_FILE_DIR:shared_api_tests>")
endif()
enable_testing()
include(GoogleTest)
# Allow DLL startup and parameterized test enumeration on busy CI runners.
gtest_discover_tests(shared_api_tests TEST_PREFIX "shared_api." DISCOVERY_TIMEOUT 30)
