# SDK fixtures are pinned original headers, never regenerated from the host.
foreach(sdk v11 partial full)
  add_library(v12_${sdk} SHARED plugin.cpp)
  target_include_directories(v12_${sdk} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk/${sdk}")
  set_target_properties(v12_${sdk} PROPERTIES PREFIX "")
endforeach()
target_compile_definitions(v12_v11 PRIVATE V12_LOCK=0 V12_FULL=0)
target_compile_definitions(v12_partial PRIVATE V12_LOCK=1 V12_FULL=0)
target_compile_definitions(v12_full PRIVATE V12_LOCK=1 V12_FULL=1)
add_library(v12_cx SHARED plugin.cpp cx_entry.cpp "${PROJECT_SOURCE_DIR}/avs_core/include/avs/cx/legacy.cpp")
target_include_directories(v12_cx PRIVATE "${PROJECT_SOURCE_DIR}/avs_core/include")
target_compile_definitions(v12_cx PRIVATE V12_LOCK=1 V12_FULL=1)
set_target_properties(v12_cx PROPERTIES PREFIX "")
add_library(v12_oldcx SHARED plugin.cpp cx_entry.cpp sdk/partial/avs/cx/legacy.cpp)
target_include_directories(v12_oldcx PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk/partial")
target_compile_definitions(v12_oldcx PRIVATE V12_LOCK=1 V12_FULL=0)
add_library(v12_missing SHARED plugin.cpp hidden_message.cpp "${PROJECT_SOURCE_DIR}/avs_core/include/avs/cx/legacy.cpp")
target_include_directories(v12_missing PRIVATE "${PROJECT_SOURCE_DIR}/avs_core/include")
target_compile_definitions(v12_missing PRIVATE V12_LOCK=1 V12_FULL=1)
add_executable(v12_tests tests.cpp c_probe.c)
target_link_libraries(v12_tests PRIVATE GTest::gtest_main AvsCoreExternal)
foreach(sdk v11 partial full cx oldcx missing)
  target_compile_definitions(v12_tests PRIVATE V12_${sdk}_PATH="$<TARGET_FILE:v12_${sdk}>")
  add_dependencies(v12_tests v12_${sdk})
endforeach()
include(GoogleTest)
gtest_discover_tests(v12_tests TEST_PREFIX "v12." PROPERTIES TIMEOUT 30)
